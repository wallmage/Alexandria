```python
@dataclass
class Finding:
    family: str          # e.g. "ledger/quantity", "fidelity/mismatch", "integrity/quotation-lost"

    klass: str           # "F" | "A"  (spec §6.10; default "F" for hard, "A" for warn unless listed)
    ids: list[str]       # claim/source ids involved, may be empty
    message: str         # threshold + actual value where applicable
    fix: str             # pinned command or closed-list imperative
    remove: str = ""     # scope-dropping command for Class F (e.g. "alx claim drop C8 --apply")
def as_text(f: Finding) -> str        # "[FAIL] ..." / "WARNING: ..." legacy string
def group(findings) -> dict[str, list[Finding]]
def render_grouped(findings, *, per_family=5) -> str   # the spec §6.7 output block

# scripts/validate_ledger.py (T2)
def collect_findings(ledger: dict, *, schema_path=None, cache_dir=None) -> list[Finding]
def claim_findings(claim: dict, ledger: dict, *, cache_dir=None) -> list[Finding]   # all families for one claim, incl. offline probes when cache_dir
def expand_claim_input(item: dict, ledger: dict, *, cache_meta: dict) -> dict          # claim-input -> full v4 claim (spec §6.4)
CLAIM_INPUT_SCHEMA = REFERENCES / "claim-input.schema.json"

# scripts/source_fidelity.py (T1)
@dataclass
class FetchResult: status: str; reason_class: str; reason: str; text: str; charset: str; url: str; final_url: str; aliases: list[str]; http_status: int|None; title: str; published: str|None; text_sha256: str
def fetch_document(url, *, cache_dir=None, refresh=False, timeout=10, deadline=None) -> FetchResult
def cache_paths(cache_dir, source_id) -> tuple[Path, Path]        # (S<n>.txt, S<n>.meta.json)
def write_cache(cache_dir, source_id, result: FetchResult) -> None
def read_cache(cache_dir, source_id) -> tuple[str, dict] | None
def probe_findings(claim, source, text) -> list[Finding]         # offline probe of one claim/source extract against text (full coverage)
def check_source_fidelity(ledger, *, fetcher=None, sample_size=8, timeout=10, cache_dir=None, deadline=None, online=False) -> dict
def issue_source_fidelity_receipt(ledger_path, receipt_path, *, policy="weighted-source-evidence-v2", cache_dir=None, deadline=None, force=False, sample_size=8, timeout=10) -> dict   # returns result incl. refreshed source ids and disclosure_required

# scripts/rewild_gate.py (T3)
def run_check(report_path, source_path, *, lang, review_note_path=None) -> list[Finding]   # all tiers, writes nothing
# run_gate(...) unchanged signature; adds keyword force=False (stale receipt overwritten without it)

# scripts/content_gate.py (T4)
def run_check(report_path, ledger_path, review_note_path=None) -> list[Finding]
# scripts/validate_report.py (T4)
def binding_findings(text, ledger) -> list[Finding]   # links ∈ url∪aliases, claim→paragraph mapping, Sources section; spec §6.7(c)
def integrity_findings(text, ledger, *, snapshot_text=None, lang) -> list[Finding]  # spec §6.7(a)
# scripts/md_to_pdf.py (T4): render_pdf(..., issue_receipt=None); CLI --issue-receipt PATH
# scripts/render_pdf_pages.py (T4): --out-dir alias; fallback chain; timeout
```
